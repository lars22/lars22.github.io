{{- /* Bilder in Artikeln: - Relative Dateinamen (Artikel aus dem Editor) werden im Artikel-Ordner gesucht und bei Bedarf auf max. 1600 px verkleinert. - Pfade mit "/" (ältere Artikel) bleiben unverändert; Breite und Höhe werden zusätzlich gesetzt, damit die Seite beim Nachladen der Fotos nicht springt. - Leerer Alt-Text (oder "image") wird durch den Artikeltitel ersetzt. */ -}} {{- $dest := .Destination -}} {{- $alt := .PlainText -}} {{- if or (eq $alt "") (eq $alt "image") -}}{{- $alt = .Page.Title -}}{{- end -}} {{- $img := "" -}} {{- if and $dest (not (hasPrefix $dest "/")) (not (hasPrefix $dest "http")) -}} {{- $img = .Page.Resources.GetMatch $dest -}} {{- end -}} {{- if $img -}} {{- if gt $img.Width 1600 -}}{{- $img = $img.Resize "1600x q85" -}}{{- end -}} {{- else -}} {{- $w := 0 -}}{{- $h := 0 -}} {{- if hasPrefix $dest "/posts/" -}} {{- with resources.Get (printf "content%s" $dest) -}}{{- $w = .Width -}}{{- $h = .Height -}}{{- end -}} {{- end -}} {{- end -}}
